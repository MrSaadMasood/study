Pattern 1: Two Pointers

When to recognize: Sorted array + find pair/triplet meeting a condition. "Two numbers that sum to target." "Remove duplicates in-place." Palindrome check. Reduce O(n²) brute force to O(n). 
// Template: two pointers from both ends
function twoPointers(arr, target) {
  let left = 0, right = arr.length - 1;
  while (left < right) {
    const sum = arr[left] + arr[right];
    if (sum === target) return [left, right];
    else if (sum < target) left++;
    else right--;
  }
  return [-1, -1];
}

// Template: two pointers same direction (slow + fast)
function removeDuplicates(nums) {
  if (!nums.length) return 0;
  let slow = 0;
  for (let fast = 1; fast < nums.length; fast++) {
    if (nums[fast] !== nums[slow]) {
      slow++;
      nums[slow] = nums[fast];
    }
  }
  return slow + 1; // length of unique portion
}

Pattern 2: Sliding Window

When to recognize: "Subarray/substring of length K" or "longest/shortest subarray satisfying condition." Variable or fixed window. Reduces O(n²) nested loops to O(n). 
// Fixed window size K
function maxSumSubarrayOfSizeK(arr, k) {
  let windowSum = 0, maxSum = 0;
  for (let i = 0; i < arr.length; i++) {
    windowSum += arr[i];
    if (i >= k - 1) {
      maxSum = Math.max(maxSum, windowSum);
      windowSum -= arr[i - (k - 1)]; // shrink from left
    }
  }
  return maxSum;
}

// Variable window — expand right, shrink left when condition violated
function lengthOfLongestSubstringKDistinct(s, k) {
  const freq = new Map();
  let left = 0, maxLen = 0;
  for (let right = 0; right < s.length; right++) {
    freq.set(s[right], (freq.get(s[right]) || 0) + 1);
    while (freq.size > k) { // shrink window
      freq.set(s[left], freq.get(s[left]) - 1);
      if (freq.get(s[left]) === 0) freq.delete(s[left]);
      left++;
    }
    maxLen = Math.max(maxLen, right - left + 1);
  }
  return maxLen;
}

Pattern 3: Fast & Slow Pointers (Floyd's Algorithm)

When to recognize: Linked list cycle detection. Middle of linked list. Start of cycle. "Does it loop?" 
// Detect cycle in linked list
function hasCycle(head) {
  let slow = head, fast = head;
  while (fast && fast.next) {
    slow = slow.next;
    fast = fast.next.next;
    if (slow === fast) return true;
  }
  return false;
}

// Find middle of linked list
function findMiddle(head) {
  let slow = head, fast = head;
  while (fast && fast.next) {
    slow = slow.next;
    fast = fast.next.next;
  }
  return slow; // slow is at middle when fast reaches end
}

Pattern 4: Binary Search & Variants

When to recognize: Sorted array + search. "Find target," "find first/last occurrence," "find minimum in rotated array," "search in answer space." O(log n) is the hint. 
// Standard binary search
function binarySearch(nums, target) {
  let left = 0, right = nums.length - 1;
  while (left <= right) {
    const mid = left + Math.floor((right - left) / 2); // avoid overflow
    if (nums[mid] === target) return mid;
    else if (nums[mid] < target) left = mid + 1;
    else right = mid - 1;
  }
  return -1;
}

// Find leftmost (first) occurrence
function findFirst(nums, target) {
  let left = 0, right = nums.length - 1, result = -1;
  while (left <= right) {
    const mid = left + Math.floor((right - left) / 2);
    if (nums[mid] === target) { result = mid; right = mid - 1; } // keep going left
    else if (nums[mid] < target) left = mid + 1;
    else right = mid - 1;
  }
  return result;
}

// Search in rotated sorted array
function searchRotated(nums, target) {
  let left = 0, right = nums.length - 1;
  while (left <= right) {
    const mid = left + Math.floor((right - left) / 2);
    if (nums[mid] === target) return mid;
    // Left half is sorted
    if (nums[left] <= nums[mid]) {
      if (target >= nums[left] && target < nums[mid]) right = mid - 1;
      else left = mid + 1;
    } else { // Right half is sorted
      if (target > nums[mid] && target <= nums[right]) left = mid + 1;
      else right = mid - 1;
    }
  }
  return -1;
}

Pattern 5: Tree BFS & DFS

When to recognize: BFS = level-by-level traversal, shortest path in unweighted graph, "level order," "right side view." DFS = explore all paths, tree structure, "validate," "path sum," "LCA." 
// BFS — Level Order Traversal
function levelOrder(root) {
  if (!root) return [];
  const result = [], queue = [root];
  while (queue.length) {
    const levelSize = queue.length;
    const level = [];
    for (let i = 0; i < levelSize; i++) {
      const node = queue.shift();
      level.push(node.val);
      if (node.left) queue.push(node.left);
      if (node.right) queue.push(node.right);
    }
    result.push(level);
  }
  return result;
}

// DFS — recursive
function maxDepth(root) {
  if (!root) return 0;
  return 1 + Math.max(maxDepth(root.left), maxDepth(root.right));
}

// DFS — iterative (with stack — avoids stack overflow on deep trees)
function inorderIterative(root) {
  const result = [], stack = [];
  let curr = root;
  while (curr || stack.length) {
    while (curr) { stack.push(curr); curr = curr.left; }
    curr = stack.pop();
    result.push(curr.val);
    curr = curr.right;
  }
  return result;
}

// Validate BST — pass min/max bounds
function isValidBST(node, min = -Infinity, max = Infinity) {
  if (!node) return true;
  if (node.val <= min || node.val >= max) return false;
  return isValidBST(node.left, min, node.val) && 
         isValidBST(node.right, node.val, max);
}

Pattern 6: Graph BFS & DFS

When to recognize: Grid problems (islands, paths). "Connected components." "Can you reach from A to B?" Topological sort for dependency ordering. 
// Graph BFS — shortest path in unweighted graph
function bfsShortestPath(graph, start, end) {
  const visited = new Set([start]);
  const queue = [[start, 0]]; // [node, distance]
  while (queue.length) {
    const [node, dist] = queue.shift();
    if (node === end) return dist;
    for (const neighbor of graph[node] || []) {
      if (!visited.has(neighbor)) {
        visited.add(neighbor);
        queue.push([neighbor, dist + 1]);
      }
    }
  }
  return -1;
}

// Number of Islands — DFS on grid
function numIslands(grid) {
  let count = 0;
  const dfs = (i, j) => {
    if (i < 0 || i >= grid.length || j < 0 || j >= grid[0].length || grid[i][j] !== '1') return;
    grid[i][j] = '0'; // mark visited (modify in place)
    dfs(i+1, j); dfs(i-1, j); dfs(i, j+1); dfs(i, j-1);
  };
  for (let i = 0; i < grid.length; i++)
    for (let j = 0; j < grid[0].length; j++)
      if (grid[i][j] === '1') { count++; dfs(i, j); }
  return count;
}

// Topological Sort (Kahn's Algorithm — BFS)
function topologicalSort(numCourses, prerequisites) {
  const inDegree = new Array(numCourses).fill(0);
  const graph = Array.from({length: numCourses}, () => []);
  for (const [a, b] of prerequisites) { graph[b].push(a); inDegree[a]++; }
  
  const queue = [];
  for (let i = 0; i < numCourses; i++) if (inDegree[i] === 0) queue.push(i);
  
  const order = [];
  while (queue.length) {
    const node = queue.shift();
    order.push(node);
    for (const neighbor of graph[node]) {
      inDegree[neighbor]--;
      if (inDegree[neighbor] === 0) queue.push(neighbor);
    }
  }
  return order.length === numCourses ? order : []; // empty = cycle exists
}

Pattern 7: Dynamic Programming

When to recognize: "Optimal substructure" — optimal solution contains optimal sub-solutions. "Overlapping subproblems" — same subproblem computed many times. "How many ways," "minimum/maximum," "can you reach." If you can define: "state = some parameters" and "transition = how to compute state from smaller states" → it's DP. 
// 1D DP: House Robber
// State: dp[i] = max money robbing up to house i
// Transition: dp[i] = max(dp[i-1], dp[i-2] + nums[i])
function rob(nums) {
  if (nums.length === 1) return nums[0];
  let prev2 = 0, prev1 = 0;
  for (const num of nums) {
    const curr = Math.max(prev1, prev2 + num);
    prev2 = prev1;
    prev1 = curr;
  }
  return prev1;
}

// 2D DP: Longest Common Subsequence
// State: dp[i][j] = LCS length of s1[0..i-1] and s2[0..j-1]
function longestCommonSubsequence(text1, text2) {
  const m = text1.length, n = text2.length;
  const dp = Array.from({length: m+1}, () => new Array(n+1).fill(0));
  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      if (text1[i-1] === text2[j-1]) dp[i][j] = dp[i-1][j-1] + 1;
      else dp[i][j] = Math.max(dp[i-1][j], dp[i][j-1]);
    }
  }
  return dp[m][n];
}

// Classic: Coin Change (minimum coins)
function coinChange(coins, amount) {
  const dp = new Array(amount + 1).fill(Infinity);
  dp[0] = 0;
  for (let i = 1; i <= amount; i++) {
    for (const coin of coins) {
      if (coin <= i) dp[i] = Math.min(dp[i], dp[i - coin] + 1);
    }
  }
  return dp[amount] === Infinity ? -1 : dp[amount];
}

Pattern 8: Heap / Priority Queue

When to recognize: "Top K elements," "Kth largest/smallest," "median of stream," "merge K sorted lists." Need to efficiently track the minimum or maximum element from a changing set. 
// JavaScript doesn't have a built-in heap — use sorted array trick for interviews
// or implement a min-heap. Mention this to the interviewer.

// Top K Frequent Elements — using frequency map + partial sort
function topKFrequent(nums, k) {
  const freq = new Map();
  for (const n of nums) freq.set(n, (freq.get(n) || 0) + 1);
  
  // Bucket sort — O(n) approach
  const buckets = new Array(nums.length + 1).fill(null).map(() => []);
  for (const [num, count] of freq) buckets[count].push(num);
  
  const result = [];
  for (let i = buckets.length - 1; i >= 0 && result.length < k; i--) {
    result.push(...buckets[i]);
  }
  return result.slice(0, k);
}

// Kth Largest — QuickSelect algorithm O(n) average
function findKthLargest(nums, k) {
  const target = nums.length - k; // kth largest = target-th smallest
  const quickSelect = (left, right) => {
    const pivot = nums[right];
    let p = left;
    for (let i = left; i < right; i++) {
      if (nums[i] <= pivot) { [nums[p], nums[i]] = [nums[i], nums[p]]; p++; }
    }
    [nums[p], nums[right]] = [nums[right], nums[p]];
    if (p === target) return nums[p];
    if (p < target) return quickSelect(p + 1, right);
    return quickSelect(left, p - 1);
  };
  return quickSelect(0, nums.length - 1);
}

Pattern 9: Stack Patterns

// Monotonic Stack — Next Greater Element pattern
function dailyTemperatures(temps) {
  const result = new Array(temps.length).fill(0);
  const stack = []; // stores indices
  for (let i = 0; i < temps.length; i++) {
    while (stack.length && temps[i] > temps[stack[stack.length-1]]) {
      const idx = stack.pop();
      result[idx] = i - idx;
    }
    stack.push(i);
  }
  return result;
}

Sorting Algorithms — Know the Trade-offs

AlgorithmBestAverageWorstSpaceStableWhen to use
Merge SortO(n log n)O(n log n)O(n log n)O(n)YesNeed stable sort, linked lists, external sort

How to talk about an algorithm in an interview:
1. State the time and space complexity immediately after presenting your solution.
2. Justify why this is optimal — "I don't think we can do better than O(n) here because we must look at every element."
3. If the interviewer asks "can you optimize?" — usually means go from O(n²) → O(n log n) → O(n).
4. Mention trade-offs: "This uses O(n) extra space — if memory is a constraint, we could use the two-pointer approach which is O(1) space."
